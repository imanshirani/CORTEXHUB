167    def launch(self):
168        """
169        Handle Launch operation.
170        """
171        # ۱. چک کردن قفل نرم‌افزار (اگر مکس مجاز نباشد، کلاً اجرا نشو)
172        if self.allowed_sw.lower() != "all" and "max" not in self.allowed_sw.lower():
173            return False, f"❌ Access Denied! This task is locked to: {self.allowed_sw.upper()}"
174
175        # ۲. پیدا کردن مسیر EXE
176        settings = QSettings("Cortex", "Pipeline")
177        exe_path = settings.value("max_path", "")
178        if not exe_path or not os.path.exists(exe_path):
179            return False, "3ds Max path is not set."
180
181        # ۳. پیدا کردن اسکریپت استارتاپ
182        current_file_path = os.path.abspath(__file__)
183        root_dir = os.path.dirname(os.path.dirname(os.path.dirname(current_file_path)))
184        script_path = os.path.join(root_dir, "plugins", "3dsmax", "startup.py")
185        
186        cmd = [exe_path]
187        if os.path.exists(script_path):
188             cmd = [exe_path, "-U", "PythonHost", script_path]
189
190        try:
191            subprocess.Popen(cmd, env=self.env)
192            return True, "3ds Max Launched!"
193        except Exception as e:
194            return False, f"Error: {e}"

224    def launch(self):
225        """
226        Handle Launch operation.
227        """
228        # 1. گرفتن مسیر فایل exe
229        settings = QSettings("Cortex", "Pipeline")
230        exe_path = settings.value("blender_path", "")
231        
232        if not exe_path or not os.path.exists(exe_path):
233            return False, "Blender path is not set. Go to Utilities tab."
234
235        # 2. پیدا کردن مسیر startup.py مخصوص بلندر
236        # مسیر: plugins/blender/startup.py
237        current_file_path = os.path.abspath(__file__)
238        root_dir = os.path.dirname(os.path.dirname(os.path.dirname(current_file_path)))
239        script_path = os.path.join(root_dir, "plugins", "blender", "startup.py")
240        
241        # 3. دستور اجرا
242        # بلندر برخلاف مکس، اسکریپت را با فلگ --python اجرا می‌کند
243        cmd = [exe_path]
244        
245        if os.path.exists(script_path):
246            print(f"Loading Blender Startup: {script_path}")
247            # دستور: blender.exe --python path/to/startup.py
248            cmd.extend(["--python", script_path])
249        else:
250            print(f"Warning: Blender startup script not found at {script_path}")
251
252        try:
253            # اجرای بلندر با متغیرهای محیطی کورتکس
254            subprocess.Popen(cmd, env=self.env)
255            return True, "Blender Launched Successfully!"
256        except Exception as e:
257            return False, str(e)
