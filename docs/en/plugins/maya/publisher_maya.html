 24    def __init__(self):
 25        """Initializes the Maya Publisher context."""
 26        self.db = DatabaseManager()
 27        self.task_id = os.environ.get("CORTEX_TASK_ID")
 28        self.work_path = os.environ.get("CORTEX_WORK_PATH") 
 29        self.user = os.environ.get("CORTEX_USER", "Unknown")
 30        self.software = "maya"

 32    def show_dialog(self):
 33        """Displays the Cortex Publish Dialog inside Maya."""
 34        if not self.task_id:
 35            QMessageBox.critical(None, "Context Error", "Please launch Maya from Cortex.")
 36            return
 37
 38        # Fake a session for the dialog
 39        class MockSession:
 40            def __init__(self, db): self.db = db
 41        session = MockSession(self.db)
 42        
 43        # Maya does not need qtmax; use the dialog directly
 44        parent = None 
 45
 46        # Detect department so output options can turn on
 47        task_data = self.db.get_task_by_id(self.task_id)
 48        dept_name = task_data[2].lower() if task_data else "general"
 49        maya_opts = self.get_maya_options(dept_name)
 50            
 51        # Screenshot the Maya viewport for the thumbnail
 52        temp_thumb = os.path.join(os.environ["TEMP"], "cortex_maya_thumb.jpg")
 53        try:
 54            # Maya command for the current viewport
 55            cmds.playblast(frame=cmds.currentTime(q=True), format="image", 
 56                           viewer=False, compression="jpg", completeFilename=temp_thumb,
 57                           widthHeight=[480, 270])
 58        except:
 59            temp_thumb = None
 60
 61        self.dialog = PublishDialog(
 62            session, 
 63            self.task_id, 
 64            parent=parent, 
 65            thumbnail_path=temp_thumb,
 66            output_config=maya_opts
 67        )
 68
 69        if self.dialog.exec():
 70            data = self.dialog.get_data()
 71            if self.validate(data):
 72                self.run_publish_process(data)

 74    def get_maya_options(self, dept_name):
 75        """Returns export options specific to Maya."""
 76        options = []
 77        options.append({'key': 'source', 'label': 'Maya Scene (.mb)', 'checked': True, 'enabled': False})
 78        
 79        if "lookdev" in dept_name:
 80            options.append({'key': 'maya_shader', 'label': 'Maya Shader Network', 'checked': True, 'enabled': True})
 81        else:
 82            options.append({'key': 'alembic', 'label': 'Alembic Cache (.abc)', 'checked': True, 'enabled': True})
 83            options.append({'key': 'export_geo', 'label': 'Export FBX (.fbx)', 'checked': True, 'enabled': True})
 84            
 85        return options

 87    def run_publish_process(self, data):
 88        """Executes the actual saving and exporting in Maya using Dynamic Paths."""
 89        import shutil 
 90        version_num = data['version']
 91        comment = data['comment']
 92        outputs = data['outputs']
 93        
 94        current_file = cmds.file(q=True, sn=True)
 95        if not current_file:
 96            QMessageBox.warning(None, "Error", "Please save your Maya file first.")
 97            return
 98
 99        dest_3d = self.db.get_publish_path(self.task_id, software=self.software, category="3d")
100        if not dest_3d:
101            QMessageBox.critical(None, "Error", "Could not resolve publish path from database.")
102            return
103
104        task_name = os.environ.get("CORTEX_TASK_NAME", "out").replace(" ", "_")
105        
106        dest_3d = os.path.join(dest_3d, task_name).replace("\\", "/")
107            
108        dest_obj = self.db.get_publish_path(self.task_id, software="obj", category="3d")
109        if dest_obj: dest_obj = os.path.join(dest_obj, task_name).replace("\\", "/")
110        
111        dest_abc = self.db.get_publish_path(self.task_id, software="abc", category="3d")
112        if dest_abc: dest_abc = os.path.join(dest_abc, task_name).replace("\\", "/")
113
114        if not os.path.exists(dest_3d): os.makedirs(dest_3d)
115        if dest_obj and not os.path.exists(dest_obj): os.makedirs(dest_obj)
116        if dest_abc and not os.path.exists(dest_abc): os.makedirs(dest_abc)
117
118        published_files = {}
119
120        try:
121            # 1. Save Scene
122            print(">> Saving Work File...")
123            cmds.file(save=True, force=True)
124
125            thumb_source = getattr(self.dialog, 'thumbnail_path', None)
126            if thumb_source and os.path.exists(thumb_source):
127                work_thumb = os.path.splitext(current_file)[0] + ".jpg"
128                try: shutil.copy2(thumb_source, work_thumb)
129                except: pass
130
131            # 2. Publish Maya File
132            if 'source' in outputs:
133                pub_path = os.path.join(dest_3d, os.path.basename(current_file)).replace("\\", "/")
134                shutil.copy2(current_file, pub_path)
135                published_files['source_maya'] = pub_path
136                print(f">> Published Maya: {pub_path}")
137
138                if thumb_source and os.path.exists(thumb_source):
139                    pub_thumb = os.path.splitext(pub_path)[0] + ".jpg"
140                    try: shutil.copy2(thumb_source, pub_thumb)
141                    except: pass
142
143            # 3. Alembic Export
144            if 'alembic' in outputs:
145                if not cmds.pluginInfo("AbcExport", q=True, loaded=True):
146                    cmds.loadPlugin("AbcExport")
147                
148                # Naming fix: same pattern as Houdini and Max (e.g. BODY_v002.abc)
149                abc_name = f"{task_name}_v{version_num:03d}.abc"
150                abc_path = os.path.join(dest_abc, abc_name).replace("\\", "/")
151                
152                print(f">> Exporting Alembic: {abc_name}")
153                
154                # Root fix: find every top-level object in the scene
155                top_nodes = cmds.ls(assemblies=True)
156                if not top_nodes:
157                    raise Exception("Scene is empty! Nothing to export.")
158                    
159                root_args = " ".join([f"-root {node}" for node in top_nodes])
160                
161                # Export using the nodes we found
162                cmds.AbcExport(j=f"-frameRange 1 1 {root_args} -file \"{abc_path}\"")
163                published_files['alembic'] = abc_path
164
165            # 4. FBX Export
166            if 'export_geo' in outputs:
167                if not cmds.pluginInfo("fbxmaya", q=True, loaded=True):
168                    cmds.loadPlugin("fbxmaya")
169                
170                # FBX naming fix
171                fbx_name = f"{task_name}_v{version_num:03d}.fbx"
172                fbx_path = os.path.join(dest_obj, fbx_name).replace("\\", "/")
173                
174                print(f">> Exporting FBX: {fbx_name}")
175                cmds.file(fbx_path, force=True, options="v=0;", typ="FBX export", pr=True, ea=True)
176                published_files['export_geo'] = fbx_path
177
178            # Register in the database
179            success, pub_id = self.db.create_publish(
180                task_id=self.task_id,
181                version=version_num,
182                comment=comment,
183                user_name=self.user,
184                thumbnail_path=thumb_source,
185                files_dict=published_files
186            )
187            
188            if success:
189                QMessageBox.information(None, "Success", f"Maya Publish v{version_num:03d} Done!")
190            else:
191                QMessageBox.critical(None, "DB Error", "Failed to register publish in DB.")
192
193        except Exception as e:
194            QMessageBox.critical(None, "Publish Failed", str(e))
195            print(f"!! Maya Publish Error: {e}")

197    def get_publish_root(self):
198        """Helper to find the asset root folder."""
199        # Same logic as Max
200        parts = self.work_path.replace("\\", "/").split("/")
201        for i, p in enumerate(parts):
202            if p.lower() == "work":
203                return "/".join(parts[:i])
204        return self.work_path

207        """Basic validation for Maya scene."""
208        # e.g. empty names or leftover history
209        return True

212        """Call the Maya material packager with an exact path"""
213        try:
214            import publisher_mat_maya
215            import importlib
216            importlib.reload(publisher_mat_maya)
217            
218            engine = self.db.get_setting("render_engine", default="Arnold") 
219            task_name = os.environ.get("CORTEX_TASK_NAME", "out").replace(" ", "_")
220            
221            # Exact path: publish/3D/lookdev/TASK_NAME
222            lookdev_base = self.db.get_publish_path(self.task_id, software="lookdev", category="3d")
223            if not lookdev_base:
224                lookdev_base = os.path.join(self.get_publish_root(), "publish", "3D", "lookdev").replace("\\", "/")
225                
226            final_mat_path = os.path.join(lookdev_base, task_name).replace("\\", "/")
