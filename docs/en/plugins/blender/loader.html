165    def refresh_scene_manager(self):
166        """Scan the scene for XRefs and check for updates"""
167        self.table_refs.setRowCount(0)
168        
169        # 1. XRef count from Max
170        xref_count = rt.xrefs.getXRefFileCount()
171        for i in range(1, xref_count + 1):
172            # MaxScript indexes start at 1
173            file_path = rt.xrefs.getXRefFile(i).filename
174            filename = os.path.basename(file_path)
175            folder = os.path.dirname(file_path)
176            
177            # 2. Parse the filename (Body_v001.max)
178            # Regex for vXXX
179            match = re.search(r"_v(\d{3})", filename)
180            
181            asset_name = filename
182            current_ver = 0
183            latest_ver = 0
184            status = "Unknown"
185            
186            if match:
187                ver_str = match.group(1)
188                current_ver = int(ver_str)
189                prefix = filename.split(f"_v{ver_str}")[0] # Body
190                
191                # 3. Look in the folder for a newer version
192                if os.path.exists(folder):
193                    files = os.listdir(folder)
194                    max_v = current_ver
195                    for f in files:
196                        if f.startswith(prefix) and f.endswith(".max") and "_v" in f:
197                            try:
198                                v = int(f.split("_v")[-1].split(".")[0])
199                                if v > max_v: max_v = v
200                            except: pass
201                    latest_ver = max_v
202            
203            # 4. Set status
204            is_outdated = latest_ver > current_ver
205            
206            # 5. Fill the table
207            row = self.table_refs.rowCount()
208            self.table_refs.insertRow(row)
209            
210            self.table_refs.setItem(row, 0, QTableWidgetItem(filename))
211            self.table_refs.setItem(row, 1, QTableWidgetItem(f"v{current_ver:03d}"))
212            
213            # Latest
214            item_lat = QTableWidgetItem(f"v{latest_ver:03d}")
215            if is_outdated:
216                item_lat.setForeground(QColor("#ff5555")) # red
217            else:
218                item_lat.setForeground(QColor("#55ff55")) # green
219            self.table_refs.setItem(row, 2, item_lat)
220            
221            # Status
222            status_text = "⚠️ Outdated" if is_outdated else "✅ OK"
223            self.table_refs.setItem(row, 3, QTableWidgetItem(status_text))
224            
225            # Action Button
226            if is_outdated:
227                btn_update = QPushButton("🚀 Update")
228                btn_update.setStyleSheet(style.BTN_UPDATE)
229                # Store the Max index and new path on the button
230                new_filename = filename.replace(f"v{current_ver:03d}", f"v{latest_ver:03d}")
231                new_full_path = os.path.join(folder, new_filename)
232                # Pass checked into the lambda to avoid TypeError
233                btn_update.clicked.connect(lambda checked=False, idx=i, path=new_full_path: self.do_update_xref(idx, path))
234                self.table_refs.setCellWidget(row, 4, btn_update)
235            else:
236                self.table_refs.setItem(row, 4, QTableWidgetItem("-"))
237
238        for obj in rt.objects:
239            mat_path = rt.getUserProp(obj, "Cortex_Mat_Source")
240            if mat_path and os.path.exists(mat_path):
241                mat_name = rt.getUserProp(obj, "Cortex_Mat_Name")
242                self.add_material_row_to_manager(obj, mat_name, mat_path)
