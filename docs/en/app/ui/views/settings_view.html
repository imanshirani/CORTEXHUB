205    def save_admin_config(self):
206        """
207        Handle Save Admin Config operation.
208        """
209        # 1. Read general values
210        new_path = self.path_input.text()
211        new_fps = self.fps_combo.currentText()
212        new_w = self.width_input.text()
213        new_h = self.height_input.text()
214        new_sw_list = self.sw_list_input.text() 
215        new_engine_list = self.engine_list_input.text()
216        
217        # 2. Read JSON values
218        proj_text = self.proj_struct_edit.toPlainText()
219        asset_text = self.asset_struct_edit.toPlainText() # added
220        shot_text = self.shot_struct_edit.toPlainText()
221
222        # Validation
223        if not new_path or not new_w or not new_h:
224            QMessageBox.warning(self, "Error", "General fields cannot be empty.")
225            return
226
227        try:
228            # Check that the entered text is valid JSON
229            json.loads(proj_text)
230            json.loads(asset_text) # added
231            json.loads(shot_text)
232        except json.JSONDecodeError as e:
233            QMessageBox.critical(self, "Syntax Error", f"Invalid JSON format!\n{e}")
234            return
235            
236        # Save all settings to the database
237        self.session.db.set_setting("server_path", new_path)
238        self.session.db.set_setting("global_fps", new_fps)
239        self.session.db.set_setting("default_width", new_w)
240        self.session.db.set_setting("default_height", new_h)
241        self.session.db.set_setting("project_structure", proj_text)
242        self.session.db.set_setting("asset_structure", asset_text)  # added
243        self.session.db.set_setting("shot_structure", shot_text)       
244        self.session.db.set_setting("allowed_softwares_list", new_sw_list)
245        self.session.db.set_setting("render_engines_list", new_engine_list)
246        
247        QMessageBox.information(self, "Success", "Admin configurations saved to database.")

303    def save_profile(self):
304        """
305        Handle Save Profile operation.
306        """
307        new_name = self.name_input.text()
308        new_pass = self.pass_input.text()
309        confirm_pass = self.confirm_input.text()
310        
311        if not new_name:
312            QMessageBox.warning(self, "Error", "Name cannot be empty.")
313            return
314
315        if new_pass and new_pass != confirm_pass:
316            QMessageBox.warning(self, "Error", "Passwords do not match!")
317            return
318        
319        success = self.session.db.update_user_profile_safe(
320            self.user.id, new_name, new_pass if new_pass else None
321        )
322
323        if success:
324            QMessageBox.information(self, "Success", "Profile updated successfully!\nChanges applied on next login.")
325            self.user.full_name = new_name
326        else:
327            QMessageBox.critical(self, "Error", "Failed to update profile.")

326            self.user.full_name = new_name
