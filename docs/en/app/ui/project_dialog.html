119            self.render_combo.setCurrentText(p.render_engine)

121    def save_project(self):
122        """
123        Handle Save Project operation.
124        """
125        # 2. Be sure to use strip so that extra space does not cause errors
126        name = self.name_input.text().strip()
127        code = self.code_input.text().upper().strip()
128        path = self.path_input.text().strip()
129        status = self.status_combo.currentText()
130        software = self.software_combo.currentText()
131        render_engine = self.render_combo.currentText()
132
133        if not name or not code or not path:
134            QMessageBox.warning(self, "Missing Data", "Please fill all fields.")
135            return
136
137        if self.project_to_edit:
138            success = self.session.db.update_project(
139                self.project_to_edit.id, name, code, path, status, render_engine, software)
140        else:
141            # Create New Project
142            success = self.session.db.create_project(name, code, path, render_engine, software)
143            
144            # --- Important part: making folders ---
145            if success:
146                # Build the full path: Root + Project Name
147                full_project_path = os.path.join(path, name)
148                
149                # Calling the folder creation function
150                FileSystemManager.create_project_structure(self.session.db, full_project_path)
151            # -----------------------------
152
153        if success:
154            self.accept()
155        else:
156            QMessageBox.critical(self, "Error", "Operation failed.")
