 54        # Core Context

143class MaxLauncher(BaseLauncher):
144    def __init__(self, session):
145        """
146        Handle   Init   operation.
147        """
148        super().__init__(session)
149        self.software_dir = "max"

151    def launch(self):
152        """
153        Handle Launch operation.
154        """
155        # 1. Software lock check (if Max is not allowed, don't run at all)
156        if self.allowed_sw.lower() != "all" and "max" not in self.allowed_sw.lower():
157            return False, f"❌ Access Denied! This task is locked to: {self.allowed_sw.upper()}"
158
159        # 2. Find the EXE path
160        settings = QSettings("Cortex", "Pipeline")
161        exe_path = settings.value("max_path", "")
162        if not exe_path or not os.path.exists(exe_path):
163            return False, "3ds Max path is not set."
164
165        # 3. Find the startup script
166        current_file_path = os.path.abspath(__file__)
167        root_dir = os.path.dirname(os.path.dirname(os.path.dirname(current_file_path)))
168        script_path = os.path.join(root_dir, "plugins", "3dsmax", "startup.py")
169        
170        cmd = [exe_path]
171        if os.path.exists(script_path):
172             cmd = [exe_path, "-U", "PythonHost", script_path]
173
174        try:
175            subprocess.Popen(cmd, env=self.env)
176            return True, "3ds Max Launched!"
177        except Exception as e:
178            return False, f"Error: {e}"

188    def launch(self):
189        # Checking the software lock from the database
190        if self.allowed_sw.lower() != "all" and "maya" not in self.allowed_sw.lower():
191            return False, f"❌ Access Denied! Task locked to: {self.allowed_sw.upper()}"
192
193        settings = QSettings("Cortex", "Pipeline")
194        exe_path = settings.value("maya_path", "") 
195        
196        if not exe_path or not os.path.exists(exe_path):
197            return False, "Maya path is not set in Settings > Utilities."
198
199        # Find the exact path of the plugins/maya folder in Cortex
200        current_file_path = os.path.abspath(__file__)
201        root_dir = os.path.dirname(os.path.dirname(os.path.dirname(current_file_path)))
202        maya_plugin_dir = os.path.join(root_dir, "plugins", "maya").replace("\\", "/")
203
204        # --- These two lines are magic! ---
205        # Add plugin path to Maya Python variable
206        existing_python_path = self.env.get("PYTHONPATH", "")
207        self.env["PYTHONPATH"] = maya_plugin_dir + os.pathsep + existing_python_path
208        # -------------------------------
209
210        try:
211            # Now open Maya with these new settings
212            subprocess.Popen([exe_path], env=self.env)
213            return True, "Maya Launched with Cortex Pipeline!"
214        except Exception as e:
215            return False, str(e)

221class HoudiniLauncher(BaseLauncher):
222    def __init__(self, session):
223        """Initializes the Houdini Launcher context."""
224        super().__init__(session)
225        self.software_dir = "houdini"
226
227    def launch(self):
228        # 1. Check access permission (department lock)
229        if self.allowed_sw.lower() != "all" and "houdini" not in self.allowed_sw.lower():
230            return False, f"❌ Access Denied! Task locked to: {self.allowed_sw.upper()}"
231
232        # 2. Reading the EXE path from local settings
233        settings = QSettings("Cortex", "Pipeline")
234        exe_path = settings.value("houdini_path", "")
235        
236        if not exe_path or not os.path.exists(exe_path):
237            return False, "Houdini path is not set in Settings > Utilities."
238
239        # 3. Find the root path of the project to address the plugins
240        current_file_path = os.path.abspath(__file__)
241        root_dir = os.path.dirname(os.path.dirname(os.path.dirname(current_file_path)))
242        houdini_plugin_dir = os.path.join(root_dir, "plugins", "houdini")
243
244        # 4. Houdini specific environment variable setting
245        # Introducing the plugin path as the root for Houdini
246        self.env["CORTEX_HOUDINI"] = houdini_plugin_dir
247        
248        # Introducing the script path to Houdini Python (to identify publisher_houdini.py)
249        scripts_path = os.path.join(houdini_plugin_dir, "scripts")
250        self.env["PYTHONPATH"] = scripts_path + os.pathsep + self.env.get("PYTHONPATH", "")
251
252        # Set JOB to access files marked $JOB in Houdini
253        self.env["JOB"] = self.work_path 
254        
255        # Introducing the path of Houdini packages (for automatic loading of the shelf and settings)
256        self.env["HOUDINI_PACKAGE_DIR"] = houdini_plugin_dir
257        
258        # Open the Houdini console to view Cortex logs
259        self.env["HOUDINI_WINDOW_CONSOLE"] = "1"
260        
261        try:
262            # 5. Running Houdini with an isolated environment (env)
263            subprocess.Popen([exe_path], env=self.env)
264            return True, "Houdini Launched with Cortex Professional Engine!"
265        except Exception as e:
266            return False, f"Houdini Launch Failed: {str(e)}"

286        # 1. Getting the path of the exe file

293        # 2. Find the startup.py path for Blender
294        # Path: plugins/blender/startup.py
295        current_file_path = os.path.abspath(__file__)
296        root_dir = os.path.dirname(os.path.dirname(os.path.dirname(current_file_path)))
297        script_path = os.path.join(root_dir, "plugins", "blender", "startup.py")
298        
299        # 3. Execution order
300        # Unlike Max, Blender runs the script with the --python flag
301        cmd = [exe_path]
302        
303        if os.path.exists(script_path):
304            print(f"Loading Blender Startup: {script_path}")
305            # Constitution: blender.exe --python path/to/startup.py
306            cmd.extend(["--python", script_path])
307        else:
308            print(f"Warning: Blender startup script not found at {script_path}")
309
310        try:
311            # Running Blender with Cortex environment variables
312            subprocess.Popen(cmd, env=self.env)
313            return True, "Blender Launched Successfully!"
